\subsection{Temporal predicates}\label{sec:app-temporal}
The order and containment predicates of \cref{sec:chaining} compare interval endpoints under a
boundary slack. For positive-duration windows $W_X=[s_X,e_X]$, the recorded executor uses a slack
equal to one quarter of the shorter window,
\begin{equation}
\begin{gathered}
\delta(A,B)=\tfrac14
  \min(e_A-s_A,e_B-s_B),\\
\rho_{\mathrm{before}}(W_A,W_B)
=
\begin{cases}
\mathsf{S}, & e_A\leq s_B+\delta(A,B),\\
\mathsf{C}, & e_B\leq s_A+\delta(A,B),\\
\mathsf{U}, & \text{otherwise}.
\end{cases}
\end{gathered}
\label{eq:temporal-before}
\end{equation}
The slack tolerates a boundary overlap no larger than $\delta$; with the quarter-window rule and
positive durations the first two cases cannot both hold. Containment uses the same slack.

\subsection{Withdrawal noise-level schedule}\label{sec:app-schedule}
The control-withdrawal probe of \cref{sec:gen-withdrawal} reports the withdrawal point by the noise
level $\sigma_{\mathrm{rel}}$ rather than the step index $i^{*}$. For $S$ steps at flow shift
$\kappa$ over $N$ training timesteps, the backbone's UniPC schedule assigns
\begin{equation}
\tilde{s}_i = \frac{(S-i)\,(1 - 1/N)}{S},
\qquad
\sigma_i = \frac{\kappa\,\tilde{s}_i}{1 + (\kappa-1)\,\tilde{s}_i},
\qquad
\sigma_{\mathrm{rel}} = \sigma_{i^{*}} ,
\label{eq:flow-sigma}
\end{equation}
where $\tilde{s}_i$ is the normalized flow-matching timestep at step $i$ and $\sigma_i$ the noise
level assigned to it. Two runs that withdraw at the same step index thus sit at different noise
levels whenever $S$ or $\kappa$ differs.

\subsection{Trajectory-agreement metrics}\label{sec:app-retention}
The control-release analysis of \cref{sec:gen-release} scores how closely a generated clip's tracked
path follows the simulation's. Both videos are tracked frame by frame; with
$p^{\mathrm{gen}}_t,p^{\mathrm{ctrl}}_t$ the tracked positions ($t\in\{0,\ldots,T-1\}$, undefined
where the tracker finds nothing) and $V$ the frames where both are defined, \emph{retention} over a
window $W$ is the per-axis Pearson correlation over $W\cap V$: $R_x(W)$ and $R_y(W)$, reported only
where $|W\cap V|$ meets a minimum count. Intersecting with $V$ drops a frame missed in either video
from both series, and $R$ near $1$ means the object moved as simulated.

The windows split at the first ground-contact frame, read off the control's own vertical path. With
$y^{\mathrm{ctrl}}$ downward and $t_{\mathrm{apex}}=\arg\min_{t\in V} y^{\mathrm{ctrl}}_t$,
\begin{equation}
t_c \;=\; \min\Big\{\, t \geq t_{\mathrm{apex}} \;:\; y^{\mathrm{ctrl}}_t \geq
   y_{\max} - \tfrac{1}{10}\big(y_{\max}-y_{\min}\big) \Big\},
\label{eq:contact-frame}
\end{equation}
$y_{\min},y_{\max}$ over the control's located frames. The \emph{airborne} window is $\{t<t_c\}$ and
\emph{after-contact} $\{t\geq t_c\}$. Two prefixed failure modes accompany retention: a
\emph{whole-clip disagreement} (full-clip horizontal $<0.7$) and a \emph{late-window disagreement}
(full-clip horizontal $<0.7$ while airborne horizontal $>0.9$, locating it after contact).

\subsection{Critic dataflow and procedure}\label{sec:app-critic-alg}
\Cref{fig:criticpipe} shows the critic dataflow of \cref{sec:critic} as a block diagram;
\cref{alg:critic} states the one-clip verdict procedure in full, annotating each step as learned or
deterministic with the equation defining it, and \cref{alg:onepass} is a condensed view. All three
summarise \cref{sec:critic} and add no new step.

\begin{algorithm}[H]
\caption{PhyReAct critic: bounded learned planning/verification embedded in deterministic compilation, scoping, and roll-up (executable summary of \cref{sec:critic}).}
\label{alg:critic}
\begin{algorithmic}[1]
\Require clip $\mathcal{V}=\{(I_t,\tau_t)\}_{t\in\mathcal{T}_{\mathcal V}}$, prompt $p$
\Ensure verdict $Y\in\{\textsc{Plausible},\textsc{Implausible},\textsc{Abstain}\}$ (or $\bot_{\mathrm{infra}}$)
\Procedure{Critic}{$\mathcal{V},p$}
  \State $(\mathcal{C},\widetilde{\Pi})\gets\Pi_\theta(p)$, \; $\mathcal{C}=\{c_i=(s_i,\phi_i)\}_{i=1}^{N_p}$, $s_i\sqsubseteq p$ \Comment{\cref{eq:claims-and-plan}; text-only, video unseen \emph{(learned)}}
  \State $G_\Pi=(\mathcal{A},\mathcal{Q},\mathcal{E},b)\gets\textsc{Compile}(\widetilde{\Pi})$, \; $a_j=(o_j,\alpha_j,\omega_j,r_j)$ \Comment{typed DAG, before any frame is read; \cref{eq:plan-dag} \emph{(det.)}}
  \If{$\neg\operatorname{Valid}(G_\Pi)$} \Comment{$\operatorname{Valid}=\operatorname{Parse}\land\operatorname{Refs}\land\operatorname{Typed}\land\operatorname{Acyclic}\land[\forall i\,\exists q\!:b(q){=}i]$; \cref{eq:plan-validity} \emph{(det.)}}
     \State $G_\Pi\gets\textsc{Fallback}(G_\Pi)$; \; \textsc{Charge}\,\&\,\textsc{Report} \Comment{invalid line$\to$general verifier, invalid term$\to$recorded sub-check; never guessed}
  \EndIf
  \For{$a_j\in\mathcal{A}$ in topological order of $G_\Pi$} \Comment{video first touched here}
     \State $\Omega_j\gets g_j(\{R_k:(a_k,a_j)\in\mathcal{E}\},\mathcal{T}_{\mathcal V})$; \; $R_j\gets F_{o_j}(\mathcal{V}|_{\Omega_j},\alpha_j)$ \Comment{obs.\ gate scope \emph{(det.)}; frozen backend, SAM~3 grounds tracks; \cref{eq:scoped-execution}}
  \EndFor
  \For{$q\in\mathcal{Q}$}
     \State $z_q\gets\psi_q(\{\mu_j:a_j\in\operatorname{Pa}(q)\})\in\mathbb{V}=\{\mathsf{S},\mathsf{C},\mathsf{U}\}$; \; negate via $\neg_3$ \Comment{$\psi_q$ det.\ for count/timing/track/transcription, learned asymmetric gate ($\mathsf{C}$ only on positive contradiction) for semantic; $\mathsf{U}$ if abstained, $\bot_{\mathrm{infra}}$ on infra error; \cref{sec:semantics}}
  \EndFor
  \For{$i\gets1$ to $N_p$}
     \State $K_i\gets\{q:b(q)=i\}$; \; $y_i\gets\mathsf{C}$ if $\exists q\!\in\!K_i\!:z_q{=}\mathsf{C}$, \; $\mathsf{S}$ if $\forall q\!\in\!K_i\!:z_q{=}\mathsf{S}$, \; else $\mathsf{U}$ \Comment{claim roll-up; \cref{eq:claim-clip-rollup} \emph{(det.)}}
  \EndFor
  \ForAll{$i$ with $y_i=\mathsf{U}$} \Comment{single post-composition step \emph{(learned)}}
     \State $y_i\gets\textsc{FallbackVerify}(c_i,\{R_j\})$
  \EndFor
  \State $Y\gets\bot_{\mathrm{infra}}$ on infra error; \; \textsc{Implausible} if $\exists i\!:y_i{=}\mathsf{C}$; \; \textsc{Plausible} if $N_p{\ge}1\land\forall i\,y_i{=}\mathsf{S}$; \; else \textsc{Abstain} \Comment{clip roll-up; \cref{eq:claim-clip-rollup} \emph{(det.)}}
  \State \Return $Y$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{\PhyReAct{}: one-clip verdict. All reasoning is compiled and validated
before any frame is read; execution only scopes declared calls, and adjudication
is deterministic.}
\label{alg:onepass}
\begin{algorithmic}[1]
\Require prompt $p$, decoded clip $\mathcal V$ (\cref{eq:video-clock})
\Ensure clip verdict $Y$, contradiction packet $\mathcal F$, provenance trace $T$
\State $(\mathcal C,\widetilde\Pi)\gets \Pi_\theta(p)$ \Comment{typed claims + surface plan, \cref{eq:claims-and-plan}}
\State $G_\Pi\gets \operatorname{Compile}(\widetilde\Pi)$ \Comment{typed graph, \cref{eq:plan-dag}}
\If{$\neg\operatorname{Valid}(G_\Pi)$} route the offending line/term to the charged fallback \Comment{\cref{eq:plan-validity}}
\EndIf
\For{$a_j$ \textbf{in} $\operatorname{TopoOrder}(G_\Pi)$} \Comment{scoped execution, \cref{eq:scoped-execution}}
  \State $\Omega_j\gets g_j(\{R_k:(a_k,a_j)\in\mathcal E\},\,\mathcal T_{\mathcal V})$ \Comment{full clip or declared local window}
  \State $R_j\gets F_{o_j}(\mathcal V|_{\Omega_j},\alpha_j)$ \Comment{frozen backend; reuse by realized signature (\cref{sec:chaining})}
  \State serialize $R_{j}$ with status and provenance \Comment{\cref{eq:evidence-record}}
\EndFor
\For{each check $q\in\mathcal Q$}
  \State $z_q\gets \psi_q(\{\mu_j:a_j\in\operatorname{Pa}(q)\})\in\{\mathsf S,\mathsf C,\mathsf U\}$ \Comment{typed resolver, \cref{sec:semantics}}
\EndFor
\For{each claim $c_i$}
  \State $y_i\gets \operatorname{compose}(\{z_q:b(q)=i\})$, negating via $\neg_3$ \Comment{\cref{eq:claim-clip-rollup}}
\EndFor
\State $Y\gets \operatorname{rollup}(\{y_i\})$; \quad $\mathcal F\gets\{(c_i,\{R_{j,i}\}_j,\Omega_i,B_i):y_i=\mathsf C\}$ \Comment{\cref{eq:claim-clip-rollup,eq:refinement-packet}}
\State \Return $(Y,\mathcal F,T)$
\end{algorithmic}
\end{algorithm}

\subsection{Benchmark details}\label{sec:app-benchmark}
Per-field coverage is in \cref{tab:flawfields}, one full record in \cref{fig:benchanatomy}, the
set's derivation in \cref{fig:benchcomp}, and the core's distributions in \cref{fig:benchdist}; the
record's fields and grouping tags are defined in \cref{sec:bench-spec}.

By machine-assigned type, the core's $306$ flaws (median two per clip) are actions that do not
happen or happen wrongly $139$; wrong counts $36$; wrong or missing objects $30$; garbled on-screen
text $25$; wrong spatial arrangement $19$; wrong ordering $17$; wrong attributes $16$; camera and
style $10$; physical motion $7$; residual $7$---not a physics benchmark in the narrow sense, with
counting, text, and arrangement together $80$.

The $150$ clips run a median $8.0$\,s ($5.0$--$10.3$), mostly $24$\,fps ($118/150$), under prompts
that are not one-line captions: median $42$ words, longest $188$, many specifying a shot list, a
count, an ordering, an on-screen string, and a sound at once. Sixty-six clips carry more than one
flaw kind, so one number per clip cannot say which a detector found. Of the $237$ time-spanned
flaws the flagged window covers a median $62\%$ of the clip, bimodally: $57$ span a quarter or less
and $95$ at least nine-tenths (\cref{fig:benchdist}), so a detector needs both a close look and a
whole-clip judgment.
